````markdown
## Ce qui tourne tout seul

Une règle écrite se contourne, un code de sortie non. Ces dispositifs
s'exécutent sans qu'on y pense — les connaître évite de les prendre pour des
pannes, et de refaire ce qu'ils font déjà.
````

```python
        for m, q in zip(moins, plus):
            assert neuf.count(m + "\n") == 1, m
            neuf = neuf.replace(m + "\n", q + "\n")
        with tempfile.TemporaryDirectory() as d:
            a = pathlib.Path(d, "a"); b = pathlib.Path(d, "b")
            a.write_text(actuel, encoding="utf-8"); b.write_text(neuf, encoding="utf-8")
            diff = subprocess.run(["diff", "-U3", str(a), str(b)], capture_output=True, text=True).stdout
            ha = subprocess.run(["git", "hash-object", str(a)], capture_output=True, text=True).stdout.strip()[:7]
            hb = subprocess.run(["git", "hash-object", str(b)], capture_output=True, text=True).stdout.strip()[:7]
        corps = diff.split("\n", 2)[2]
        entete = part.split("\n")[1]
        mode = re.match(r"index \w+\.\.\w+( \d+)?", entete).group(1) or ""
        out.append(f"diff --git a/CLAUDE.md b/CLAUDE.md\nindex {ha}..{hb}{mode}\n--- a/CLAUDE.md\n+++ b/CLAUDE.md\n{corps}")
    p.write_text("".join(out), encoding="utf-8")
```

**La vérification.** Le script ci-dessous relève 218 faits chiffrés dans

**La relecture sans contexte** : sept constats, tous corrigés.
- `engine/tests/outillage_documente.rs` parcourait le journal, où
  l'archive nomme chaque dispositif : en retirer un de `CLAUDE.md` laissait
  le test vert. Il écarte désormais les couches gelées de la mémoire, comme
  le test du banc, par leur chemin. Un cas sur un arbre fabriqué le garde ;
  le témoin tombe dans les deux sens, et retirer un dispositif de

  aucune valeur attendue ne change.

- Les exemples de « c'est plus fort » et de « technique standard »

```python
import re, sys, pathlib
# Usage, depuis la racine du dépôt :
#   python3 chiffres.py [--sans-claude] <ancien texte> [<ancien texte> ...]
# Relève les faits chiffrés de l'ancien texte et dit lesquels ne vivent ni
# dans CLAUDE.md, ni dans memory/ hors de l'entrée d'archive J-2026-11-17-Q,
# ni dans tools/README.md. Un verdict écrit à un chiffre après la virgule
# (« +10,7 ± 3,4 ») vit aussi s'il est écrit plus précis ailleurs et qu'il
# s'y arrondit (« +10,66 ± 3,43 »).
ARCHIVE = "J-2026-11-17-Q"
args = sys.argv[1:]
sans_claude = "--sans-claude" in args
args = [a for a in args if a != "--sans-claude"]
def norm(t):
    t = t.replace("\u202f", " ").replace("\u00a0", " ").replace("\u2212", "-")
    return re.sub(r"([×÷]) (\d)", r"\1\2", t)
avant = norm("\n".join(pathlib.Path(a).read_text(encoding="utf-8") for a in args))
reste = ""
for p in pathlib.Path("memory").rglob("*.md"):
    t = p.read_text(encoding="utf-8")
    if f"## {ARCHIVE} " in t:
        i = t.index(f"## {ARCHIVE} "); j = t.find("\n## J-", i + 5); j = len(t) if j < 0 else j
        t = t[:i] + t[j:]
    reste += norm(t)
faits = set(re.findall(r"[+-]?\d+(?:,\d+)? ± \d+(?:,\d+)?", avant))
faits |= set(re.findall(r"[×÷]\d+(?:,\d+)?", avant)) | set(re.findall(r"\d+(?:,\d+)? %", avant))
faits |= set(re.findall(r"\d{1,3}(?: \d{3})+(?: (?:coups|parties|positions|nœuds|n/s|mutants))?", avant))
faits |= set(re.findall(r"[+-]\d+,\d+", avant))
vivant = reste + norm(pathlib.Path("tools/README.md").read_text(encoding="utf-8"))
if not sans_claude:
    vivant += norm(pathlib.Path("CLAUDE.md").read_text(encoding="utf-8"))
from decimal import Decimal, ROUND_HALF_UP
num = lambda s: Decimal(s.replace(",", "."))
rnd = lambda x, d: abs(x).quantize(Decimal(1).scaleb(-d), rounding=ROUND_HALF_UP).copy_sign(x)
precis = [(m.group(1), m.group(2)) for m in re.finditer(r"([+-]?\d+,\d+) ± (\d+,\d+)", vivant)]
def arrondi_vivant(f):
    m = re.fullmatch(r"([+-]?\d+,(\d+)) ± (\d+,(\d+))", f)
    if not m:
        return False
    v, dv, e, de = m.group(1), len(m.group(2)), m.group(3), len(m.group(4))
    for pv, pe in precis:
        if pv[0] in "+-" and v[0] in "+-" and pv[0] != v[0]:
            continue
        if rnd(num(pv), dv) == num(v) and rnd(num(pe), de) == num(e):
            return True
    return False
seuls = sorted(f for f in faits if f not in vivant and not arrondi_vivant(f))
print(f"{len(faits)} faits chiffrés, {len(faits) - len(seuls)} vivants hors de l'archive")
for f in seuls:
    print("  seulement dans l'ancien texte :", f)
```
